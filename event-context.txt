<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\HasOne;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use LogicException;

class Event extends Model
{
    public const REVIEW_DRAFT = 'draft';
    public const REVIEW_SUBMITTED = 'submitted';
    public const REVIEW_APPROVED = 'approved';
    public const REVIEW_REJECTED = 'rejected';

    public const SALES_OPEN = 'open';
    public const SALES_FROZEN = 'frozen';
    public const SALES_SOLD_OUT = 'sold_out';

    // Everything else (ownership, review, activation, toggles, freeze) is set
    // server-side through explicit actions, never from a request payload.
    protected $fillable = [
        'title', 'description', 'category', 'restrictions', 'refund_policy',
        'starts_at', 'ends_at', 'currency', 'is_free',
    ];

    protected function casts(): array
    {
        return [
            'starts_at' => 'datetime',
            'ends_at' => 'datetime',
            'is_free' => 'boolean',
            'reviewed_at' => 'datetime',
            'creation_fee_paid_at' => 'datetime',
            'activated_at' => 'datetime',
            'made_public_at' => 'datetime',
            'made_bookable_at' => 'datetime',
            'frozen_at' => 'datetime',
        ];
    }

    public function organizer(): BelongsTo
    {
        return $this->belongsTo(Organizer::class);
    }

    public function floorPlan(): BelongsTo
    {
        return $this->belongsTo(FloorPlan::class);
    }

    public function financialTerms(): HasOne
    {
        return $this->hasOne(EventFinancialTerms::class);
    }

    public function seats(): HasMany
    {
        return $this->hasMany(EventSeat::class);
    }

    public function getVenueAttribute(): ?Venue
    {
        return $this->floorPlan?->venue;
    }

    /** The ownership check from the roles matrix (0b): organizers only ever see their own events. */
    public function scopeOwnedBy(Builder $query, Organizer $organizer): Builder
    {
        return $query->where('organizer_id', $organizer->id);
    }

    public static function generateSlug(string $title): string
    {
        return Str::slug($title).'-'.Str::lower(Str::random(6));
    }

    public function isApproved(): bool
    {
        return $this->review_status === self::REVIEW_APPROVED;
    }

    public function isActivated(): bool
    {
        return $this->activated_at !== null;
    }

    /** Free events pay no creation fee, so there is nothing to wait for. */
    public function creationFeeSettled(): bool
    {
        return $this->is_free || $this->creation_fee_paid_at !== null;
    }

    /** Make Public: Approved + fee settled + Admin-Activated. Not gated by floor plan status. */
    public function canBeMadePublic(): bool
    {
        return $this->isApproved() && $this->creationFeeSettled() && $this->isActivated();
    }

    /** Make Bookable: everything Public needs, plus a ready floor plan. */
    public function canBeMadeBookable(): bool
    {
        return $this->canBeMadePublic() && $this->floorPlan?->isReady() === true;
    }

    /**
     * Visible to visitors. Re-evaluates the gates every time, so an Admin
     * deactivating the event hides it even if the organizer's toggle is still on.
     */
    public function isVisibleToPublic(): bool
    {
        return $this->made_public_at !== null && $this->canBeMadePublic();
    }

    public function isPurchasable(): bool
    {
        return $this->made_bookable_at !== null
            && $this->sales_status === self::SALES_OPEN
            && $this->canBeMadeBookable();
    }

    /**
     * Copy the floor plan's seats into event_seats (the per-event snapshot).
     *
     * - The floor plan must be ready: a custom plan has no seats until it is built,
     *   so for those events the snapshot happens after Admin marks it ready.
     * - Re-snapshotting is only allowed while the event is a draft. After that the
     *   layout is locked, and seats with sales can never be moved or deleted.
     */
    public function snapshotSeats(): int
    {
        $plan = $this->floorPlan;

        if (! $plan || ! $plan->isReady()) {
            throw new LogicException('The floor plan must be ready before its seats can be snapshotted.');
        }

        if ($this->seats()->exists() && $this->review_status !== self::REVIEW_DRAFT) {
            throw new LogicException('Seats were already snapshotted and the layout is locked after submission.');
        }

        return DB::transaction(function () use ($plan) {
            $this->seats()->delete();

            $now = now();
            $rows = $plan->seats()->get()->map(fn (FloorPlanSeat $seat) => [
                'event_id' => $this->id,
                'floor_plan_seat_id' => $seat->id,
                'seat_key' => $seat->seat_key,
                'section' => $seat->section,
                'row_label' => $seat->row_label,
                'seat_number' => $seat->seat_number,
                'created_at' => $now,
                'updated_at' => $now,
            ])->all();

            foreach (array_chunk($rows, 500) as $chunk) {
                EventSeat::insert($chunk);
            }

            return count($rows);
        });
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use RuntimeException;

class EventFinancialTerms extends Model
{
    protected $table = 'event_financial_terms';

    // Overrides are an admin action behind the override_financial_terms permission,
    // so nothing here is mass-assignable.
    protected $fillable = [];

    protected function casts(): array
    {
        return [
            'reserve_enabled' => 'boolean',
            'overridden_at' => 'datetime',
        ];
    }

    public function event(): BelongsTo
    {
        return $this->belongsTo(Event::class);
    }

    public static function defaultsFor(string $currency): array
    {
        $schedule = config("bookcat.fees.{$currency}");

        if (! $schedule) {
            throw new RuntimeException("No fee schedule configured for {$currency}.");
        }

        return $schedule;
    }

    /** Copy the platform defaults onto the event. Fully free events carry no fees at all. */
    public static function createDefaultsFor(Event $event): self
    {
        $defaults = self::defaultsFor($event->currency);
        $free = $event->is_free;

        $terms = new self();
        $terms->forceFill([
            'event_id' => $event->id,
            'creation_fee_cents' => $free ? 0 : $defaults['creation_fee_cents'],
            'ticket_fee_flat_cents' => $free ? 0 : $defaults['ticket_fee_flat_cents'],
            'ticket_fee_bps' => $free ? 0 : $defaults['ticket_fee_bps'],
            'processing_fee_bps' => $free ? 0 : $defaults['processing_fee_bps'],
            'minimum_charge_cents' => $free ? 0 : $defaults['minimum_charge_cents'],
            'payout_delay_business_days' => $defaults['payout_delay_business_days'],
        ])->save();

        return $terms;
    }
}

namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\Relations\HasMany;

class FloorPlan extends Model
{
    public const KIND_TEMPLATE = 'template';
    public const KIND_CUSTOM = 'custom';

    public const STATUS_REQUESTED = 'requested';
    public const STATUS_IN_PROGRESS = 'in_progress';
    public const STATUS_READY = 'ready';

    // venue, owner, kind, status and ready_* are set server-side only.
    protected $fillable = ['name'];

    protected function casts(): array
    {
        return ['ready_at' => 'datetime'];
    }

    public function venue(): BelongsTo
    {
        return $this->belongsTo(Venue::class);
    }

    public function owner(): BelongsTo
    {
        return $this->belongsTo(Organizer::class, 'owner_organizer_id');
    }

    public function seats(): HasMany
    {
        return $this->hasMany(FloorPlanSeat::class);
    }

    public function isReady(): bool
    {
        return $this->status === self::STATUS_READY;
    }

    /**
     * Floor plans an organizer may pick for an event: ready ones that are
     * either library templates or that organizer's own custom builds.
     * This is the ownership check from the roles matrix (section 0b) for floor plans.
     */
    public function scopeSelectableBy(Builder $query, Organizer $organizer): Builder
    {
        return $query->where('status', self::STATUS_READY)
            ->where(function (Builder $q) use ($organizer) {
                $q->whereNull('owner_organizer_id')
                    ->orWhere('owner_organizer_id', $organizer->id);
            });
    }

            return;

        $plan = FloorPlan::where('kind', FloorPlan::KIND_TEMPLATE)

            'origin' => 'sales_assisted',
            'application_status' => Organizer::STATUS_APPROVED,
            'reviewed_by' => $super->id,
            'reviewed_at' => now(),
        ])->save();

        $event = new Event([
            'title' => 'Demo Concert',
            'description' => 'Seeded demo event for local development.',
            'category' => 'Entertaining',
            'starts_at' => now()->addMonths(2),
            'currency' => 'CAD',
            'is_free' => false,

            'organizer_id' => $organizer->id,
            'created_by' => $account->id,
            'floor_plan_id' => $plan->id,
            'slug' => Event::generateSlug('Demo Concert'),

        EventFinancialTerms::createDefaultsFor($event);

        // Snapshot and price while the event is still a draft

            'reviewed_by' => $super->id,
            'reviewed_at' => now(),
            'creation_fee_paid_at' => now(),
            'activated_at' => now(),
            'activated_by' => $super->id,
            'made_public_at' => now(),
            'made_bookable_at' => now(),
        ])->save();
    }

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class FloorPlanSeat extends Model
{
    public $timestamps = false;

    protected $fillable = ['floor_plan_id', 'seat_key', 'section', 'row_label', 'seat_number'];

    public function floorPlan(): BelongsTo
    {
        return $this->belongsTo(FloorPlan::class);
    }

    /**
     * The one place seat keys are built, so the SVG tracer, API and Redis
     * layer can never disagree. Example: makeKey('Orch', 'c', 12) => "ORCH-C-12"
     */
    public static function makeKey(string $section, ?string $row, string|int $number): string
    {
        $clean = fn ($value) => strtoupper(preg_replace('/[^A-Za-z0-9]/', '', (string) $value));

        return implode('-', array_filter(
            [$clean($section), $clean($row ?? ''), $clean($number)],
            fn ($part) => $part !== ''
        ));
    }
}
